% cvitems outside a \cventry cell (see resume/experience.tex): restore the

  {\vspace{2mm}\fontsize{9pt}{1em}\bodyfontlight\color{text}\emergencystretch=1em\begin{cvitems}}
  {\end{cvitems}\vspace{6mm}}

%%% Renderers for the section lists in shared/content.tex

%%% \cventry ends in horizontal mode, and the space after it is part of the tuned

%%% the entry's paragraph, or the next entry joins it and overflows the margin.

  \cventry{#1}{#2}{#3}{#4}{}
  \ifblank{#5}{\par}{\begin{cvbullets}#5\end{cvbullets}}}
