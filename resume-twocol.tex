% !TeX program = xelatex
%%% Two-column layout experiment, built on AltaCV (https://github.com/liantze/AltaCV).
%%% The one-column awesome-cv version in resume.tex is unchanged; both build side by side.
\documentclass[10pt,letterpaper,ragged2e,withhyper]{altacv}

\geometry{left=1.3cm,right=1.3cm,top=1.3cm,bottom=1.6cm,columnsep=1.1cm}

%%% AltaCV expects the document to load paracol itself
\usepackage{paracol}

%%% Reuse the fonts already vendored for awesome-cv so both layouts look related
\setmainfont{Roboto}[
  Path=fonts/,
  Extension=.ttf,
  UprightFont=*-Regular,
  BoldFont=*-Bold,
  ItalicFont=*-Italic,
  BoldItalicFont=*-BoldItalic
]

%%% Match awesome-cv's accent (awesome-red) for visual continuity
\definecolor{AwesomeRed}{HTML}{DC3522}
\definecolor{DarkNight}{HTML}{131A28}
\definecolor{SlateGrey}{HTML}{2E2E2E}
\definecolor{LightGrey}{HTML}{666666}

\colorlet{name}{DarkNight}
\colorlet{tagline}{AwesomeRed}
\colorlet{heading}{AwesomeRed}
\colorlet{headingrule}{AwesomeRed}
\colorlet{subheading}{DarkNight}
\colorlet{accent}{AwesomeRed}
\colorlet{emphasis}{SlateGrey}
\colorlet{body}{LightGrey}

%%% Narrow columns plus long hyphenated compounds (HIPAA-regulated, SHA-pinned)
%%% leave TeX no good break; a little emergency stretch clears the last overfulls.
\setlength{\emergencystretch}{1em}

\renewcommand{\cvItemMarker}{{\small\textbullet}}
\renewcommand{\cvRatingMarker}{\faCircle}

%%% Upstream \cvtag is sized for a wider sidebar; at 0.34 columnratio the longer
%%% tags overrun the column. Same box, smaller type and tighter padding.
%%% The trailing \penalty0 plus glue is the separator between tags: tikz can swallow
%%% the whitespace after \node's semicolon, which leaves a row of tags with no legal
%%% breakpoint and forces an overfull line rather than a wrap.
\renewcommand{\cvtag}[1]{%
  \tikz[baseline]\node[anchor=base,draw=body!30,rounded corners,inner xsep=0.55ex,%
    inner ysep=0.6ex,text height=1.4ex,text depth=.25ex]{\small#1};%
  \penalty0\hskip 0.3em plus 0.2em minus 0.1em\relax
}

%%% A labelled block of \cvtag boxes, wrapping over as many lines as it needs.
%%% flushleft rather than a bare \raggedright: inside paracol the plain switch did not
%%% survive, leaving justified glue that overfills the column instead of wrapping.
\newcommand{\skillgroup}[2]{%
  \textbf{\small\color{emphasis}#1}\par
  %%% \hbadness: with ragged-right tag rows the leftover space goes to the right
  %%% margin, so an "underfull" line is invisible rather than a defect worth reporting.
  \begin{flushleft}\sloppy\hbadness=10000 #2\end{flushleft}
  \smallskip
}

\input{shared/content.tex}

%%% Renderers for the section lists in shared/content.tex.
%%% \forcsvlist does not expand its list, so the \skills* macro is expanded first and
%%% each comma-separated item becomes its own tag. The space after each tag keeps the
%%% spacing the hand-written tag rows had.
\newcommand{\skilltag}[1]{\cvtag{#1} }
\newcommand{\rSkill}[2]{%
  \skillgroup{#1}{\expandafter\forcsvlist\expandafter\skilltag\expandafter{#2}}}
%%% Jobs are separated by a divider, so every job but the first draws one first.
\newif\iffirstjob
\newcommand{\rJob}[5]{%
  \iffirstjob\firstjobfalse\else\divider\par\fi
  \cvevent{#1}{#2}{#4}{#3}
  \ifblank{#5}{}{\begin{itemize}#5\end{itemize}}\par}
\newcommand{\rEducation}[4]{\cvevent{#1}{#2}{#4}{#3}\par}
\newcommand{\rCertification}[3]{\cvevent{#1}{#2}{#3}{}\par}

  \ifdefempty{\myLinkedin}{}{\linkedin{\myLinkedin}}

%%% Sidebar on the left, main content on the right
%%% Wide enough for the longest single \cvtag ("Supply-Chain Hardening"): a tag is one
%%% unbreakable box, so a narrower sidebar cannot wrap it and spills it into the main
%%% column instead.
\columnratio{0.40}
\begin{paracol}{2}

%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%

%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%

}

\end{paracol}
