%%% Upstream \cvtag is sized for a wider sidebar; at 0.34 columnratio the longer
%%% tags overrun the column. Same box, smaller type and tighter padding.
%%% The trailing \penalty0 plus glue is the separator between tags: tikz can swallow
%%% the whitespace after \node's semicolon, which leaves a row of tags with no legal
%%% breakpoint and forces an overfull line rather than a wrap.
\renewcommand{\cvtag}[1]{%
  \tikz[baseline]\node[anchor=base,draw=body!30,rounded corners,inner xsep=0.55ex,%
    inner ysep=0.6ex,text height=1.4ex,text depth=.25ex]{\small#1};%
  \penalty0\hskip 0.3em plus 0.2em minus 0.1em\relax
}

%%% A labelled block of \cvtag boxes, wrapping over as many lines as it needs.
%%% flushleft rather than a bare \raggedright: inside paracol the plain switch did not
%%% survive, leaving justified glue that overfills the column instead of wrapping.
\newcommand{\skillgroup}[2]{%
  \textbf{\small\color{emphasis}#1}\par
  %%% \hbadness: with ragged-right tag rows the leftover space goes to the right
  %%% margin, so an "underfull" line is invisible rather than a defect worth reporting.
  \begin{flushleft}\sloppy\hbadness=10000 #2\end{flushleft}
  \smallskip
}
