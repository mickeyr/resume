%%% Template-neutral resume content.
%%%
%%% Each layout candidate needs the same words in a different macro set. Keeping the
%%% prose here means an edit to a bullet lands in every candidate at once, and an A/B
%%% comparison is comparing layouts rather than drifting copies of the text.
%%%
%%% Bullets are emitted as \rItem{...}, which each template redefines before use.
%%% This exists because moderncv typesets a \cventry description argument inside a box,
%%% and a box cannot break across pages: a nested 14-item itemize there gets pushed
%%% whole to the next page and leaves most of a page blank. Mapping \rItem onto a
%%% page-breakable per-class construct (\cvlistitem, or a plain \item at top level)
%%% avoids that.
\providecommand{\rItem}[1]{\item #1}
