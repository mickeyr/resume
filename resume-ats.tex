% !TeX program = xelatex
%%% ATS layout: single column, no graphics, no tables, no icons, plain headings.
%%% This file exists to be converted, not admired: `make resume-ats.docx` runs it
%%% through pandoc, and pandoc can only see structure a plain article class emits.
%%% The fancier candidates (awesome-cv, altacv, moderncv) encode headings, dates and
%%% contact details in class-specific macros and boxes that pandoc drops on the floor.
\documentclass[11pt,letterpaper]{article}
\usepackage[margin=0.9in]{geometry}
\usepackage{enumitem}
\setlist[itemize]{leftmargin=*,itemsep=2pt,topsep=2pt}
\pagestyle{empty}

\input{shared/content.tex}

%%% pandoc does not implement etoolbox's conditionals and silently drops them with
%%% everything inside, which would leave the docx with no sections. TeX ignores these
%%% (content.tex already loaded etoolbox); pandoc gets the non-empty branch, so an
%%% empty field or list shows as a blank line or bare heading in the docx only.
\providecommand{\ifdefempty}[3]{#3}
\providecommand{\ifblank}[3]{#3}

%%% Renderers for the section lists in shared/content.tex
\newcommand{\rSkill}[2]{\item #1: #2}
\newcommand{\rJob}[5]{%
  \subsection*{#1, #2}
  #4, #3
  \ifblank{#5}{\par}{\begin{itemize}#5\end{itemize}}}
\newcommand{\rEducation}[4]{#1, #2, #3 --- #4\par}
\newcommand{\rCertification}[3]{#1, #2 --- #3\par}

\begin{document}
% Ragged right: justification cannot stretch lines full of unbreakable
% slash-joined tech names ("Terraform/OpenTofu"), and overflows the margin instead.
\raggedright

\begin{center}
{\LARGE \myName}\\[2pt]
\myPosition\\[2pt]
% One contact item per line: pandoc drops \textbar, and a bare | typesets as an
% em dash in the default text font, so separators are a line break instead.
\myEmail
\ifdefempty{\myPhone}{}{\\\myPhone}
\\\myCity
\ifdefempty{\myGithub}{}{\\github.com/\myGithub}
\ifdefempty{\myLinkedin}{}{\\linkedin.com/in/\myLinkedin}
\end{center}

\section*{Summary}
\mySummary

\ifdefempty{\mySkills}{}{%
\section*{Skills}
\begin{itemize}
\mySkills
\end{itemize}
}

\ifdefempty{\myExperience}{}{%
\section*{Experience}
\myExperience
}

\ifdefempty{\myEducation}{}{%
\section*{Education}
\myEducation
}

\ifdefempty{\myCertifications}{}{%
\section*{Certifications}
\myCertifications
}

\end{document}
